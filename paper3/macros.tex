% macros.tex -- Paper 3: statistics lookup, anonymization switch, figure, notice and notation macros.
%
% Numbers from the campaign are never typed in. Every result is written as \stat{key} (running
% text) or \tstat{key} (table cells), where key is a dotted stats.json key such as
% x1.h100.baseline.sf300.energy_total.estimate. The \stat machinery is Paper 1's
% (../paper/macros.tex): a stats file generated/stats.tex with one \defstat{key}{value} per key
% and \statsfinaltrue. In final mode a missing key is a hard LaTeX error; in draft mode (no
% generated/stats.tex, as now) the key is printed in a yellow TODO box, and table cells print a
% red TODO (the cell's key pattern is in the table's caption comment).
%
% The keys are paths of analyze_paper3.py's flattened stats.json (checked on a synthetic tree,
% 2026-11-23); "make stats-check STATS=<out>/stats.json" lists any the analysis output lacks.
%
% Pre-lock method-validation numbers (probes P1-P7, G8 probe; design sections 2.1-2.4, 4) are
% typed directly: they are synthetic-load QA values fixed before the lock, not campaign outcomes.

\makeatletter
\newif\ifstatsfinal
\statsfinalfalse
\newcommand{\defstat}[2]{\expandafter\def\csname stat@#1\endcsname{#2}}
% Breakable (seqsplit) so that long keys do not overfill two-column lines in draft builds.
\newcommand{\statplaceholder}[1]{%
  {\color{red}\tiny\ttfamily[TODO:\expandafter\seqsplit\expandafter{\detokenize{#1}}]}}
\newcommand{\stat@missing}[1]{%
  \PackageError{stats}{Missing statistic '#1' in generated/stats.tex}%
    {Re-run the stats generator; the analysis output lacks this key.}}
\newcommand{\stat}[1]{%
  \ifcsname stat@#1\endcsname
    \csname stat@#1\endcsname
  \else
    \ifstatsfinal\stat@missing{#1}\else\statplaceholder{#1}\fi
  \fi}
% Compact form for table cells.
\newcommand{\tstat}[1]{%
  \ifcsname stat@#1\endcsname
    \csname stat@#1\endcsname
  \else
    \ifstatsfinal\stat@missing{#1}\else\textcolor{red}{\tiny TODO}\fi
  \fi}
% "estimate [ci_low, ci_high]" for a ratio prefix such as x1.h100.baseline.sf300.energy_total
\newcommand{\ratioci}[1]{%
  \stat{#1.estimate}\,{\footnotesize[\stat{#1.ci_low},\,\stat{#1.ci_high}]}}
% "estimate x (95% CI ci_low--ci_high)" for running text
\newcommand{\ratiotext}[1]{%
  \stat{#1.estimate}$\times$ (95\%~CI \stat{#1.ci_low}--\stat{#1.ci_high})}
% A long hash value (sha256), line-breakable.
\newcommand{\hashstat}[1]{%
  \ifcsname stat@#1\endcsname
    {\ttfamily\expandafter\expandafter\expandafter\seqsplit\expandafter\expandafter
      \expandafter{\csname stat@#1\endcsname}}%
  \else\stat{#1}\fi}
% Table form of \ratioci: one red TODO in draft mode instead of three.
\newcommand{\tratioci}[1]{%
  \ifcsname stat@#1.estimate\endcsname
    \tstat{#1.estimate}\,{\scriptsize[\tstat{#1.ci_low},\,\tstat{#1.ci_high}]}%
  \else
    \ifstatsfinal\stat@missing{#1.estimate}\else\textcolor{red}{\tiny TODO}\fi
  \fi}
\makeatother

% Two value files:
%  generated/stats.tex  -- written by "make stats STATS=<out>/stats.json" (tools/stats_json_to_tex.py): one

%                          analysis's own number format; \syntheticstatstrue for a synthetic tree. Its presence
%                          switches final mode on.
%  record.tex           -- tracked, hand-kept \defstat lines for values from committed records, not campaign outcomes:
%                          amendment A7 (amend.a7.date, .commit, .sha256) and launch-record facts

\newif\ifsyntheticstats
\IfFileExists{record.tex}{\input{record.tex}}{}
\IfFileExists{generated/stats.tex}{\input{generated/stats.tex}\statsfinaltrue}{}

% Compact contrast cell for wide tables: "estimate^class". The class is the analysis's word
% (design 5.3; stats.tex escapes "_" as "\_"), shown as G (gpu_better), C (cpu_better),
% = (no_difference), ? (inconclusive) or -- (not_estimable); one red TODO in draft mode.
\makeatletter
\edef\xcls@gpu{\detokenize{gpu\_better}}
\edef\xcls@cpu{\detokenize{cpu\_better}}
\edef\xcls@nod{\detokenize{no\_difference}}
\edef\xcls@inc{\detokenize{inconclusive}}
\newcommand{\xclass}[1]{%
  \ifcsname stat@#1\endcsname
    \edef\xcls@v{\expandafter\expandafter\expandafter\detokenize\expandafter\expandafter
      \expandafter{\csname stat@#1\endcsname}}%
    \ifx\xcls@v\xcls@gpu G\else\ifx\xcls@v\xcls@cpu C\else
    \ifx\xcls@v\xcls@nod =\else\ifx\xcls@v\xcls@inc ?\else --\fi\fi\fi\fi
  \else
    \ifstatsfinal\stat@missing{#1}\else\textcolor{red}{\tiny TODO}\fi
  \fi}
\newcommand{\xcell}[1]{%
  \ifcsname stat@#1.estimate\endcsname
    \tstat{#1.estimate}\textsuperscript{\xclass{#1.class}}%
  \else
    \ifstatsfinal\stat@missing{#1.estimate}\else\textcolor{red}{\tiny TODO}\fi
  \fi}
\makeatother

% ---------------------------------------------------------------- result wording
% Phrases that hold whichever way a result falls. They read the analysis's own words from the stats
% file: the class (gpu_better, cpu_better, no_difference, inconclusive, not_estimable), "n/a" for a
% missing value, and the sign of Delta*. An unexpected class prints as it is, never as another class.
% In draft mode each prints one TODO box; in final mode a missing key is an error (via \stat).
\makeatletter
\edef\xcls@nes{\detokenize{not\_estimable}}
\edef\res@na{\detokenize{n/a}}
\edef\res@minus{\detokenize{-}}
\def\res@first#1#2\relax{\def\res@f{#1}}
\newcommand{\res@set}[1]{\edef\res@v{\expandafter\expandafter\expandafter\detokenize
  \expandafter\expandafter\expandafter{\csname stat@#1\endcsname}}}
% \resna{key}{if the value is n/a}{otherwise}
\newcommand{\resna}[3]{\ifcsname stat@#1\endcsname\res@set{#1}\ifx\res@v\res@na #2\else #3\fi
  \else #3\fi}
% \resneg{key}{if the value is negative}{otherwise}
\newcommand{\resneg}[3]{\ifcsname stat@#1\endcsname\res@set{#1}%
  \expandafter\res@first\res@v\relax\relax\ifx\res@f\res@minus #2\else #3\fi\else #3\fi}
% \rtext{prefix}: "1.23x (95% CI 1.10--1.40)", "1.23x (one round, no interval)" or "not estimable"
\newcommand{\rtext}[1]{\ifcsname stat@#1.estimate\endcsname
  \resna{#1.estimate}{not estimable}{\resna{#1.ci_low}{\stat{#1.estimate}$\times$ (one round, no
  interval)}{\stat{#1.estimate}$\times$ (95\%~CI \stat{#1.ci_low}--\stat{#1.ci_high})}}%
  \else\stat{#1.estimate}\fi}
% \cls{prefix}: the class of a time or cost cell, in the words of Section 6.4
\newcommand{\cls}[1]{\ifcsname stat@#1.class\endcsname\res@set{#1.class}%
  \ifx\res@v\xcls@gpu GPU better\else\ifx\res@v\xcls@cpu CPU better\else
  \ifx\res@v\xcls@nod no difference\else\ifx\res@v\xcls@inc inconclusive\else
  \ifx\res@v\xcls@nes not estimable\else\stat{#1.class}\fi\fi\fi\fi\fi
  \else\stat{#1.class}\fi}
% \ecls{prefix}: the class of a cross-class energy cell, in the words allowed under N1 (Section 4.6)
\newcommand{\ecls}[1]{\ifcsname stat@#1.class\endcsname\res@set{#1.class}%
  \ifx\res@v\xcls@gpu GPU better at the measured scopes (no node-level claim)\else
  \ifx\res@v\xcls@cpu CPU better at the measured scopes, so under \NOne{} the GPU engine uses more
  node energy\else
  \ifx\res@v\xcls@nod inside the $\pm$10\% band at the measured scopes (no node-level claim)\else
  \ifx\res@v\xcls@inc inconclusive\else
  \ifx\res@v\xcls@nes not estimable\else\stat{#1.class}\fi\fi\fi\fi\fi
  \else\stat{#1.class}\fi}
% \dstar{prefix}: Delta* in Wh per suite run and W over the GPU suite time, with what its sign means
\newcommand{\dstar}[1]{\ifcsname stat@#1.wh_per_suite_run.mean\endcsname
  \resna{#1.wh_per_suite_run.mean}{not estimable}{\stat{#1.wh_per_suite_run.mean}\,Wh per suite run
  (\resna{#1.wh_per_suite_run.ci_low}{one round, no interval}{95\%~CI
  \stat{#1.wh_per_suite_run.ci_low} to \stat{#1.wh_per_suite_run.ci_high}};
  \stat{#1.w_over_gpu_suite_time.mean}\,W over the GPU suite time), \resneg{#1.wh_per_suite_run.mean}%
  {negative, so under \NOne{} the GPU node uses more energy}{non-negative, so the GPU node would use
  less node energy only if its unmeasured excess stayed below it; no such claim is made}}%
  \else\stat{#1.wh_per_suite_run.mean}\fi}
\makeatother
